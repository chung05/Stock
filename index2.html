// 1. ⚠️ 這行還是要保留，因為這是你 Supabase 後端的絕對路徑 const BACKEND_URL = 'https://fekesirsqjbkrgaibrjf.supabase.co/functions/v1/rapid-service'; // 2. ✨ 既然在同一個專案下，直接寫檔案路徑就可以了！不用管帳號和專案名 const DATA_URL = './data/ai_analysis.jsonl'; const PROMPT_URL = './data/prompt.json'; async function fetchAndAnalyze() { const btn = document.getElementById('btn'); const output = document.getElementById('output'); btn.disabled = true; output.innerText = "正在讀取同專案下的 ai_analysis.jsonl 與 prompt.json..."; try { // 瀏覽器會自動依據 index.html 的位置，去旁邊的 data 資料夾抓檔案 const [dataRes, promptRes] = await Promise.all([ fetch(DATA_URL), fetch(PROMPT_URL) ]); if (!dataRes.ok || !promptRes.ok) { throw new Error("無法讀取檔案，請確認 data 資料夾內是否有這兩個檔案。"); } const rawDataText = await dataRes.text(); const promptJson = await promptRes.json(); output.innerText = "成功讀取檔案！正在上傳至 Supabase 後端並等待 Gemini 分析..."; // 打包傳給後端（這段保持不變） const response = await fetch(BACKEND_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ uploadData: rawDataText, customPrompt: promptJson.prompt }) }); const resultData = await response.json(); if (resultData.report) { output.innerText = resultData.report; } else { output.innerText = "出錯了：" + (resultData.error || "未知錯誤"); } } catch (err) { output.innerText = "執行失敗：" + err.message; } finally { btn.disabled = false; } }
