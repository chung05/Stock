const DATA_URL = './data/ai_analysis.jsonl'; // 🌟 1. 改成讀取 .txt 檔案 const PROMPT_URL = './data/prompt.txt'; async function fetchAndAnalyze() { const btn = document.getElementById('btn'); const output = document.getElementById('output'); btn.disabled = true; output.innerText = "正在讀取同專案下的籌碼數據與專業 Prompt..."; try { const [dataRes, promptRes] = await Promise.all([ fetch(DATA_URL), fetch(PROMPT_URL) // 瀏覽器會去抓 prompt.txt ]); if (!dataRes.ok || !promptRes.ok) { throw new Error("無法讀取檔案，請確認 data 資料夾內是否有這兩個檔案。"); } const rawDataText = await dataRes.text(); // 🌟 2. 修改：因為是純文字檔，這裡改用 .text() 而不是 .json() const myLongPrompt = await promptRes.text(); output.innerText = "成功讀取法人分析模型！正在上傳至 Supabase 後端讓 Gemini 分析..."; // 打包傳給後端（後端完全不需要修改，因為它本來就是接收字串） const response = await fetch(BACKEND_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ uploadData: rawDataText, customPrompt: myLongPrompt // 🌟 這裡直接把整大串文字傳過去 }) }); const resultData = await response.json(); if (resultData.report) { output.innerText = resultData.report; } else { output.innerText = "出錯了：" + (resultData.error || "未知錯誤"); } } catch (err) { output.innerText = "執行失敗：" + err.message; } finally { btn.disabled = false; } }
